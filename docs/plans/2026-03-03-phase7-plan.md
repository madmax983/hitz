use std::net::SocketAddr;

use hitz_hal::Hypervisor;
use hyper::service::service_fn;
use hyper_util::rt::TokioIo;
use tokio::net::windows::named_pipe::ServerOptions;
use tokio::net::TcpListener;

use crate::router;
use crate::vm_manager::VmManager;

/// Run the named pipe HTTP server, with an optional TCP listener.
///
/// The pipe listener always runs. If `tcp_addr` is `Some`, a TCP
/// listener runs in parallel on the given address. Both share the
/// same `route()` handler. Blocks until canceled.
pub async fn run_server<H>(
    pipe_path: &str,
    tcp_addr: Option<SocketAddr>,
    manager: VmManager<H>,
) -> Result<(), std::io::Error>
where
    H: Hypervisor + Send + Sync + 'static,
{
    // Optionally start TCP listener.
    if let Some(addr) = tcp_addr {
        let tcp_manager = manager.clone();
        drop(tokio::spawn(async move {
            if let Err(e) = run_tcp_listener(addr, tcp_manager).await {
                tracing::error!("TCP listener error: {e}");
            }
        }));
        tracing::info!("TCP listener on {addr}");
    }

    // Named pipe listener (always on).
    let mut first = true;
    loop {
        let server = ServerOptions::new()
            .first_pipe_instance(first)
            .create(pipe_path)?;
        first = false;

        server.connect().await?;
        tracing::debug!("client connected to {pipe_path}");

        let mgr = manager.clone();
        drop(tokio::spawn(async move {
            let io = TokioIo::new(server);
            let service = service_fn(move |req| {
                let m = mgr.clone();
                async move { router::route(req, &m).await }
            });

            if let Err(e) = hyper::server::conn::http1::Builder::new()
                .serve_connection(io, service)
                .await
            {
                tracing::error!("connection error: {e}");
            }
        }));
    }
}

async fn run_tcp_listener<H>(
    addr: SocketAddr,
    manager: VmManager<H>,
) -> Result<(), std::io::Error>
where
    H: Hypervisor + Send + Sync + 'static,
{
    let listener = TcpListener::bind(addr).await?;

    loop {
        let (stream, peer) = listener.accept().await?;
        tracing::debug!("TCP client connected from {peer}");

        let mgr = manager.clone();
        drop(tokio::spawn(async move {
            let io = TokioIo::new(stream);
            let service = service_fn(move |req| {
                let m = mgr.clone();
                async move { router::route(req, &m).await }
            });

            if let Err(e) = hyper::server::conn::http1::Builder::new()
                .serve_connection(io, service)
                .await
            {
                tracing::error!("TCP connection error from {peer}: {e}");
            }
        }));
    }
}
```

// crates/hitz-vmm/src/serial_buf.rs

//! Shared ring buffer for serial output capture.
//!
//! `SerialBuf` implements `Write` and stores output in a fixed-size ring.
//! `SerialReader` provides async reading from an independent cursor position.

use std::io::{self, Write};
use std::sync::{Arc, Mutex};

use tokio::sync::Notify;

/// Default ring buffer capacity (64 KB).
const DEFAULT_CAPACITY: usize = 64 * 1024;

/// Shared state inside the ring buffer.
struct Inner {
    /// Fixed-size ring buffer.
    buf: Vec<u8>,
    /// Next write position (wraps around).
    write_pos: usize,
    /// Total bytes written (monotonic, never wraps).
    total_written: u64,
    /// True once the writer has been closed.
    closed: bool,
}

/// A ring buffer that captures serial output.
///
/// Implements `Write` so it can be passed to `SerialDevice<SerialBuf>`.
/// Cloning gives a handle to the same underlying buffer.
#[derive(Clone)]
pub struct SerialBuf {
    inner: Arc<Mutex<Inner>>,
    notify: Arc<Notify>,
}

impl SerialBuf {
    /// Create a new ring buffer with the default capacity (64 KB).
    #[must_use]
    pub fn new() -> Self {
        Self::with_capacity(DEFAULT_CAPACITY)
    }

    /// Create a new ring buffer with the given capacity in bytes.
    #[must_use]
    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            inner: Arc::new(Mutex::new(Inner {
                buf: vec![0u8; capacity],
                write_pos: 0,
                total_written: 0,
                closed: false,
            })),
            notify: Arc::new(Notify::new()),
        }
    }

    /// Create a reader starting from the current write position.
    ///
    /// The reader will only see data written after this call.
    #[must_use]
    pub fn reader(&self) -> SerialReader {
        let read_pos = self
            .inner
            .lock()
            .map(|inner| inner.total_written)
            .unwrap_or(0);

        SerialReader {
            inner: self.inner.clone(),
            notify: self.notify.clone(),
            read_pos,
        }
    }

    /// Mark the buffer as closed. Wakes all waiting readers.
    pub fn close(&self) {
        if let Ok(mut inner) = self.inner.lock() {
            inner.closed = true;
        }
        self.notify.notify_waiters();
    }
}

impl Default for SerialBuf {
    fn default() -> Self {
        Self::new()
    }
}

impl Write for SerialBuf {
    fn write(&mut self, data: &[u8]) -> io::Result<usize> {
        let mut inner = self
            .inner
            .lock()
            .map_err(|e| io::Error::new(io::ErrorKind::Other, e.to_string()))?;

        let cap = inner.buf.len();
        for &byte in data {
            inner.buf[inner.write_pos % cap] = byte;
            inner.write_pos = (inner.write_pos + 1) % cap;
        }
        inner.total_written += data.len() as u64;
        drop(inner);

        self.notify.notify_waiters();
        Ok(data.len())
    }

    fn flush(&mut self) -> io::Result<()> {
        Ok(())
    }
}

/// Reads serial output from a `SerialBuf` at an independent cursor.
///
/// Multiple readers can exist simultaneously, each tracking their own
/// position in the stream.
pub struct SerialReader {
    inner: Arc<Mutex<Inner>>,
    notify: Arc<Notify>,
    /// Absolute position in the total_written stream.
    read_pos: u64,
}

impl SerialReader {
    /// Read available bytes, waiting if none are available.
    ///
    /// Returns `None` when the writer is closed and all data has been read.
    pub async fn read_chunk(&mut self) -> Option<Vec<u8>> {
        loop {
            {
                let inner = self
                    .inner
                    .lock()
                    .ok()?;

                if self.read_pos < inner.total_written {
                    let available = inner.total_written - self.read_pos;
                    let cap = inner.buf.len();

                    // If we're more than a full buffer behind, skip to latest.
                    if available > cap as u64 {
                        self.read_pos = inner.total_written - cap as u64;
                    }

                    let to_read = (inner.total_written - self.read_pos) as usize;
                    let mut result = Vec::with_capacity(to_read);

                    // Read position in the ring buffer.
                    let start = (inner.write_pos + cap - to_read) % cap;
                    if start + to_read <= cap {
                        result.extend_from_slice(&inner.buf[start..start + to_read]);
                    } else {
                        // Wraps around.
                        result.extend_from_slice(&inner.buf[start..]);
                        result.extend_from_slice(&inner.buf[..to_read - (cap - start)]);
                    }

                    self.read_pos = inner.total_written;
                    return Some(result);
                }

                if inner.closed {
                    return None;
                }
            }

            self.notify.notified().await;
        }
    }
}

struct ChannelBody {

impl Body for ChannelBody {

async fn handle_serial<H>(
    id: &str,
    manager: &VmManager<H>,
) -> Result<Response<BoxBody<Bytes, Infallible>>, DaemonError>
where
    H: Hypervisor + Send + Sync + 'static,
{
    let mut reader = manager.serial_reader(id)?;
    let (tx, rx) = tokio::sync::mpsc::channel::<Bytes>(64);

    drop(tokio::spawn(async move {
        while let Some(chunk) = reader.read_chunk().await {
            if tx.send(Bytes::from(chunk)).await.is_err() {
                break; // client disconnected
            }
        }
    }));

    Ok(Response::builder()
        .status(StatusCode::OK)
        .header("content-type", "application/octet-stream")
        .header("transfer-encoding", "chunked")
        .body(ChannelBody { rx }.boxed())
        .expect("build serial response"))
}
```

// crates/hitz-net/src/ethernet.rs

//! Ethernet frame parsing and construction.
//!
//! Handles the L2↔L3 translation between virtio-net (Ethernet frames)
//! and WinTun (raw IP packets).

/// Ethernet header size in bytes.
pub const ETH_HEADER_LEN: usize = 14;
/// EtherType for IPv4.
pub const ETHERTYPE_IPV4: u16 = 0x0800;
/// EtherType for IPv6.
pub const ETHERTYPE_IPV6: u16 = 0x86DD;
/// EtherType for ARP.
pub const ETHERTYPE_ARP: u16 = 0x0806;
/// ARP operation: request.
pub const ARP_OP_REQUEST: u16 = 1;
/// ARP operation: reply.
pub const ARP_OP_REPLY: u16 = 2;

/// Parsed Ethernet header.
#[derive(Debug, Clone)]
pub struct EthHeader {
    /// Destination MAC address.
    pub dst_mac: [u8; 6],
    /// Source MAC address.
    pub src_mac: [u8; 6],
    /// EtherType (IPv4, IPv6, ARP, etc.).
    pub ethertype: u16,
}

/// Parse an Ethernet header from a frame.
///
/// Returns `None` if the frame is too short.
#[must_use]
pub fn parse_eth_header(frame: &[u8]) -> Option<EthHeader> {
    if frame.len() < ETH_HEADER_LEN {
        return None;
    }
    let mut dst_mac = [0u8; 6];
    let mut src_mac = [0u8; 6];
    dst_mac.copy_from_slice(&frame[0..6]);
    src_mac.copy_from_slice(&frame[6..12]);
    let ethertype = u16::from_be_bytes([frame[12], frame[13]]);
    Some(EthHeader {
        dst_mac,
        src_mac,
        ethertype,
    })
}

/// Build an Ethernet frame by prepending a header to an IP packet.
#[must_use]
pub fn build_eth_frame(src_mac: [u8; 6], dst_mac: [u8; 6], ethertype: u16, payload: &[u8]) -> Vec<u8> {
    let mut frame = Vec::with_capacity(ETH_HEADER_LEN + payload.len());
    frame.extend_from_slice(&dst_mac);
    frame.extend_from_slice(&src_mac);
    frame.extend_from_slice(&ethertype.to_be_bytes());
    frame.extend_from_slice(payload);
    frame
}

/// Strip the Ethernet header from a frame, returning the IP payload.
///
/// Returns `None` if the frame is too short.
#[must_use]
pub fn strip_eth_header(frame: &[u8]) -> Option<&[u8]> {
    if frame.len() > ETH_HEADER_LEN {
        Some(&frame[ETH_HEADER_LEN..])
    } else {
        None
    }
}

/// Detect EtherType from an IP packet's version field.
#[must_use]
pub fn ethertype_from_ip(packet: &[u8]) -> u16 {
    match packet.first().map(|b| b >> 4) {
        Some(4) => ETHERTYPE_IPV4,
        Some(6) => ETHERTYPE_IPV6,
        _ => ETHERTYPE_IPV4, // fallback
    }
}

/// Build an ARP reply in response to an ARP request.
///
/// Returns `None` if the request isn't a valid ARP request for `target_ip`.
#[must_use]
pub fn build_arp_reply(
    request_frame: &[u8],
    gateway_mac: [u8; 6],
    gateway_ip: [u8; 4],
) -> Option<Vec<u8>> {
    // ARP packet starts after Ethernet header.
    if request_frame.len() < ETH_HEADER_LEN + 28 {
        return None;
    }
    let arp = &request_frame[ETH_HEADER_LEN..];

    // Check: hardware type = Ethernet (1), protocol = IPv4 (0x0800),
    // operation = request (1).
    let hw_type = u16::from_be_bytes([arp[0], arp[1]]);
    let proto_type = u16::from_be_bytes([arp[2], arp[3]]);
    let operation = u16::from_be_bytes([arp[6], arp[7]]);

    if hw_type != 1 || proto_type != ETHERTYPE_IPV4 || operation != ARP_OP_REQUEST {
        return None;
    }

    // Target IP (bytes 24..28 in ARP packet).
    let target_ip_in_request = &arp[24..28];
    if target_ip_in_request != gateway_ip {
        return None;
    }

    // Sender MAC (bytes 8..14) and sender IP (bytes 14..18).
    let mut sender_mac = [0u8; 6];
    sender_mac.copy_from_slice(&arp[8..14]);
    let mut sender_ip = [0u8; 4];
    sender_ip.copy_from_slice(&arp[14..18]);

    // Build ARP reply.
    let mut reply = Vec::with_capacity(ETH_HEADER_LEN + 28);
    // Ethernet header: dst=sender, src=gateway.
    reply.extend_from_slice(&sender_mac);
    reply.extend_from_slice(&gateway_mac);
    reply.extend_from_slice(&ETHERTYPE_ARP.to_be_bytes());
    // ARP: hw_type=1, proto=0x0800, hw_len=6, proto_len=4, op=reply(2).
    reply.extend_from_slice(&[0x00, 0x01, 0x08, 0x00, 0x06, 0x04]);
    reply.extend_from_slice(&ARP_OP_REPLY.to_be_bytes());
    // Sender = gateway.
    reply.extend_from_slice(&gateway_mac);
    reply.extend_from_slice(&gateway_ip);
    // Target = original sender.
    reply.extend_from_slice(&sender_mac);
    reply.extend_from_slice(&sender_ip);

    Some(reply)
}

/// Generate a random MAC address with the locally-administered bit set.
#[must_use]
pub fn random_mac() -> [u8; 6] {
    let mut mac = [0u8; 6];
    // Use simple pseudo-random from system time.
    let seed = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_nanos())
        .unwrap_or(42);
    for (i, byte) in mac.iter_mut().enumerate() {
        *byte = ((seed >> (i * 8)) & 0xFF) as u8;
    }
    mac[0] = (mac[0] & 0xFE) | 0x02; // locally administered, unicast
    mac
}

/// Parse a MAC address string like "AA:BB:CC:DD:EE:FF".
pub fn parse_mac(s: &str) -> Result<[u8; 6], String> {
    let parts: Vec<&str> = s.split(':').collect();
    if parts.len() != 6 {
        return Err(format!("invalid MAC format: {s}"));
    }
    let mut mac = [0u8; 6];
    for (i, part) in parts.iter().enumerate() {
        mac[i] = u8::from_str_radix(part, 16)
            .map_err(|_| format!("invalid MAC byte: {part}"))?;
    }
    Ok(mac)
}

/// Parse an IP address from a CIDR string like "192.168.100.1/24".
/// Returns (ip_bytes, prefix_len).
pub fn parse_cidr(s: &str) -> Result<([u8; 4], u8), String> {
    let parts: Vec<&str> = s.splitn(2, '/').collect();
    if parts.len() != 2 {
        return Err(format!("invalid CIDR: {s}"));
    }
    let ip: std::net::Ipv4Addr = parts[0]
        .parse()
        .map_err(|e| format!("invalid IP: {e}"))?;
    let prefix: u8 = parts[1]
        .parse()
        .map_err(|e| format!("invalid prefix: {e}"))?;
    Ok((ip.octets(), prefix))
}

// crates/hitz-devices/src/virtio/net.rs

//! Virtio network device backend.
//!
//! Implements [`VirtioBackend`] for network I/O. Communicates with a
//! dedicated I/O thread via crossbeam channels.
//!
//! Device ID: 1 (virtio network device).
//! Queues: 0 = RX (host→guest), 1 = TX (guest→host).

use std::collections::VecDeque;

use crossbeam_channel::{Receiver, Sender};
use hitz_hal::GuestMemAccess;

use crate::virtio::mmio_transport::VirtioBackend;
use crate::virtio::queue::VirtQueue;

/// virtio-net header size (no merge buffers, no hash).
const VIRTIO_NET_HDR_SIZE: usize = 12;

/// `VIRTIO_NET_F_MAC` feature bit — device has a given MAC address.
const VIRTIO_NET_F_MAC: u64 = 1 << 5;

/// RX queue index.
const RX_QUEUE: u16 = 0;
/// TX queue index.
const TX_QUEUE: u16 = 1;

/// Virtio network device.
///
/// The device communicates with a host-side I/O thread via channels:
/// - `tx_sender`: guest→host Ethernet frames (from TX queue kicks)
/// - `rx_receiver`: host→guest Ethernet frames (delivered via `poll_rx`)
pub struct VirtioNetDevice {
    /// Guest MAC address.
    mac: [u8; 6],
    /// Channel to send guest TX frames to the I/O thread.
    tx_sender: Sender<Vec<u8>>,
    /// Channel to receive host RX frames from the I/O thread.
    rx_receiver: Receiver<Vec<u8>>,
    /// Frames waiting to be delivered to the guest RX queue.
    rx_pending: VecDeque<Vec<u8>>,
}

impl VirtioNetDevice {
    /// Create a new virtio-net device.
    ///
    /// Returns the device and the channel endpoints for the I/O thread:
    /// - `tx_receiver`: I/O thread reads guest TX frames from here
    /// - `rx_sender`: I/O thread sends host RX frames here
    #[must_use]
    pub fn new(mac: [u8; 6]) -> (Self, Receiver<Vec<u8>>, Sender<Vec<u8>>) {
        let (tx_sender, tx_receiver) = crossbeam_channel::unbounded();
        let (rx_sender, rx_receiver) = crossbeam_channel::unbounded();

        let device = Self {
            mac,
            tx_sender,
            rx_receiver,
            rx_pending: VecDeque::new(),
        };

        (device, tx_receiver, rx_sender)
    }

    /// Process a TX queue kick — read frames from guest and send to I/O thread.
    fn process_tx(&mut self, queue: &mut VirtQueue, mem: &dyn GuestMemAccess) {
        while let Some(chain) = queue.pop_chain(mem) {
            let head = chain.head_index();
            let mut frame = Vec::new();

            for desc in chain {
                if !desc.is_device_writable {
                    // Device-readable = guest data (TX).
                    let mut buf = vec![0u8; desc.len as usize];
                    if mem.read_guest(desc.gpa.as_u64(), &mut buf).is_ok() {
                        frame.extend_from_slice(&buf);
                    }
                }
            }

            // Strip the virtio-net header (first 12 bytes), send raw Ethernet frame.
            if frame.len() > VIRTIO_NET_HDR_SIZE {
                let eth_frame = frame[VIRTIO_NET_HDR_SIZE..].to_vec();
                let _ = self.tx_sender.send(eth_frame);
            }

            queue.push_used(mem, head, 0);
        }
    }

    /// Try to deliver pending RX frames to the guest.
    fn deliver_rx(&mut self, queue: &mut VirtQueue, mem: &dyn GuestMemAccess) -> bool {
        let mut delivered = false;

        while let Some(frame) = self.rx_pending.front() {
            if let Some(chain) = queue.pop_chain(mem) {
                let head = chain.head_index();
                let frame = self.rx_pending.pop_front().expect("just peeked");

                // Build packet: virtio-net header (all zeros) + Ethernet frame.
                let mut packet = vec![0u8; VIRTIO_NET_HDR_SIZE];
                packet.extend_from_slice(&frame);

                let mut offset = 0;
                let mut total_written = 0u32;
                for desc in chain {
                    if desc.is_device_writable && offset < packet.len() {
                        let write_len = (desc.len as usize).min(packet.len() - offset);
                        let _ = mem.write_guest(
                            desc.gpa.as_u64(),
                            &packet[offset..offset + write_len],
                        );
                        offset += write_len;
                        total_written += write_len as u32;
                    }
                }

                queue.push_used(mem, head, total_written);
                delivered = true;
            } else {
                break; // No available RX buffers.
            }
        }

        delivered
    }
}

impl VirtioBackend for VirtioNetDevice {
    fn device_id(&self) -> u32 {
        1 // virtio network
    }

    fn device_features(&self) -> u64 {
        VIRTIO_NET_F_MAC
    }

    fn queue_count(&self) -> usize {
        2 // RX + TX
    }

    fn process_queue(&mut self, queue_idx: u16, queue: &mut VirtQueue, mem: &dyn GuestMemAccess) {
        match queue_idx {
            TX_QUEUE => self.process_tx(queue, mem),
            RX_QUEUE => {
                // Guest added new RX buffers. Try to deliver pending frames.
                let _ = self.deliver_rx(queue, mem);
            }
            _ => {}
        }
    }

    fn poll_rx(&mut self, queues: &mut [&mut VirtQueue], mem: &dyn GuestMemAccess) -> bool {
        // Drain channel into pending queue.
        while let Ok(frame) = self.rx_receiver.try_recv() {
            self.rx_pending.push_back(frame);
        }

        if self.rx_pending.is_empty() {
            return false;
        }

        // Try to deliver to RX queue (index 0).
        if let Some(rx_queue) = queues.first_mut() {
            self.deliver_rx(rx_queue, mem)
        } else {
            false
        }
    }

    fn read_config(&self, offset: u64, data: &mut [u8]) {
        // Config space: MAC address at offset 0 (6 bytes).
        #[allow(clippy::cast_possible_truncation)]
        let start = offset as usize;
        if start < 6 {
            let end = (start + data.len()).min(6);
            let len = end - start;
            data[..len].copy_from_slice(&self.mac[start..end]);
        }
    }

    fn write_config(&mut self, _offset: u64, _data: &[u8]) {
        // MAC is read-only.
    }
}

//!
//! Bridges virtio-net Ethernet frames to WinTun IP packets.

use std::sync::Arc;
use std::sync::atomic::{AtomicBool, Ordering};
use std::thread;
use std::time::Duration;

use crossbeam_channel::{Receiver, Sender};

use crate::ethernet;

/// Handle to a running network I/O thread.
pub struct NetIoHandle {
    thread: Option<thread::JoinHandle<()>>,
}

impl NetIoHandle {
    /// Wait for the I/O thread to finish.
    pub fn join(mut self) {
        if let Some(handle) = self.thread.take() {
            let _ = handle.join();
        }
    }
}

impl Drop for NetIoHandle {
    fn drop(&mut self) {
        if let Some(handle) = self.thread.take() {
            let _ = handle.join();
        }
    }
}

/// Start the network I/O thread.
///
/// Creates a WinTun adapter, configures the host-side IP, and spawns a
/// thread that bridges Ethernet frames between the virtio-net device
/// channels and the WinTun adapter.
///
/// # Arguments
///
/// * `adapter_name` — Name for the WinTun adapter (e.g. "hitz-vm1")
/// * `host_ip` — Host-side IP in CIDR notation (e.g. "192.168.100.1/24")
/// * `guest_mac` — Guest MAC address
/// * `gateway_mac` — Fake gateway MAC address
/// * `gateway_ip` — Gateway IP bytes (e.g. [192, 168, 100, 1])
/// * `tx_receiver` — Receives Ethernet frames from guest TX queue
/// * `rx_sender` — Sends Ethernet frames to guest RX queue
/// * `stop_flag` — Checked each iteration; thread exits when true
pub fn start_net_io(
    adapter_name: &str,
    host_ip: &str,
    guest_mac: [u8; 6],
    gateway_mac: [u8; 6],
    gateway_ip: [u8; 4],
    tx_receiver: Receiver<Vec<u8>>,
    rx_sender: Sender<Vec<u8>>,
    stop_flag: Arc<AtomicBool>,
) -> Result<NetIoHandle, NetIoError> {
    // Load wintun.dll.
    let wintun = unsafe { wintun::load() }
        .map_err(|e| NetIoError::WinTun(format!("failed to load wintun.dll: {e}")))?;

    // Create adapter.
    let adapter = wintun::Adapter::create(&wintun, "Hitz", adapter_name, None)
        .map_err(|e| NetIoError::WinTun(format!("failed to create adapter: {e}")))?;

    // Configure host-side IP via netsh.
    configure_host_ip(adapter_name, host_ip)?;

    // Start WinTun session.
    let session = Arc::new(
        adapter
            .start_session(wintun::MAX_RING_CAPACITY)
            .map_err(|e| NetIoError::WinTun(format!("failed to start session: {e}")))?,
    );

    let session_clone = session.clone();
    let name = adapter_name.to_string();
    let thread = thread::Builder::new()
        .name(format!("hitz-net-{name}"))
        .spawn(move || {
            net_io_loop(
                &session_clone,
                &tx_receiver,
                &rx_sender,
                guest_mac,
                gateway_mac,
                gateway_ip,
                &stop_flag,
            );
        })
        .map_err(|e| NetIoError::Io(e))?;

    Ok(NetIoHandle {
        thread: Some(thread),
    })
}

fn net_io_loop(
    session: &wintun::Session,
    tx_receiver: &Receiver<Vec<u8>>,
    rx_sender: &Sender<Vec<u8>>,
    guest_mac: [u8; 6],
    gateway_mac: [u8; 6],
    gateway_ip: [u8; 4],
    stop_flag: &AtomicBool,
) {
    loop {
        if stop_flag.load(Ordering::Relaxed) {
            break;
        }

        // Process guest TX frames.
        while let Ok(frame) = tx_receiver.try_recv() {
            if let Some(hdr) = ethernet::parse_eth_header(&frame) {
                match hdr.ethertype {
                    ethernet::ETHERTYPE_ARP => {
                        // Handle ARP locally.
                        if let Some(reply) =
                            ethernet::build_arp_reply(&frame, gateway_mac, gateway_ip)
                        {
                            let _ = rx_sender.send(reply);
                        }
                    }
                    ethernet::ETHERTYPE_IPV4 | ethernet::ETHERTYPE_IPV6 => {
                        // Strip Ethernet header, send raw IP to WinTun.
                        if let Some(ip_packet) = ethernet::strip_eth_header(&frame) {
                            if let Ok(mut wt_packet) =
                                session.allocate_send_packet(ip_packet.len().try_into().unwrap_or(u16::MAX))
                            {
                                wt_packet.bytes_mut().copy_from_slice(ip_packet);
                                session.send_packet(wt_packet);
                            }
                        }
                    }
                    _ => {
                        tracing::debug!(ethertype = hdr.ethertype, "ignoring unknown ethertype");
                    }
                }
            }
        }

        // Receive host RX packets from WinTun.
        match session.try_receive() {
            Ok(Some(packet)) => {
                let ip_data = packet.bytes().to_vec();
                let ethertype = ethernet::ethertype_from_ip(&ip_data);
                let frame =
                    ethernet::build_eth_frame(gateway_mac, guest_mac, ethertype, &ip_data);
                let _ = rx_sender.send(frame);
            }
            Ok(None) => {
                // No packet available.
                thread::sleep(Duration::from_millis(1));
            }
            Err(e) => {
                tracing::error!("WinTun receive error: {e}");
                break;
            }
        }
    }
}

/// Configure the host-side IP address on the WinTun adapter via netsh.
fn configure_host_ip(adapter_name: &str, cidr: &str) -> Result<(), NetIoError> {
    let (ip, prefix) = ethernet::parse_cidr(cidr)
        .map_err(|e| NetIoError::Config(e))?;

    let ip_str = format!("{}.{}.{}.{}", ip[0], ip[1], ip[2], ip[3]);
    let mask = prefix_to_mask(prefix);

    let output = std::process::Command::new("netsh")
        .args([
            "interface",
            "ip",
            "set",
            "address",
            adapter_name,
            "static",
            &ip_str,
            &mask,
        ])
        .output()
        .map_err(NetIoError::Io)?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(NetIoError::Config(format!(
            "netsh failed: {stderr}"
        )));
    }

    Ok(())
}

/// Convert a CIDR prefix length to a dotted-decimal subnet mask.
fn prefix_to_mask(prefix: u8) -> String {
    let mask: u32 = if prefix >= 32 {
        0xFFFF_FFFF
    } else {
        !((1u32 << (32 - prefix)) - 1)
    };
    format!(
        "{}.{}.{}.{}",
        (mask >> 24) & 0xFF,
        (mask >> 16) & 0xFF,
        (mask >> 8) & 0xFF,
        mask & 0xFF,
    )
}

    #[error("I/O error: {0}")]
    Io(#[from] std::io::Error),
    /// Configuration error.
    #[error("config error: {0}")]
    Config(String),
}

#[test]
#[ignore]
fn phase7_vm_manager_serial_streaming() {
    // Tests that serial output from boot_and_run reaches SerialBuf.
    // Uses the Phase 2 "Hello" ELF which prints to COM1.
    let rt = tokio::runtime::Builder::new_current_thread()
        .enable_all()
        .build()
        .expect("tokio runtime");

    rt.block_on(async {
        let hv = Arc::new(WhpHypervisor::new().expect("WHP"));
        let manager = hitz_daemon::VmManager::new(hv);

        let config = VmConfig {
            kernel_path: elf_path("hello_serial.elf"),
            initramfs_path: None,
            disk_path: None,
            ram_mib: 128,
            cmdline: None,
            net: None,
        };

        manager
            .create_vm("serial-test".into(), config)
            .expect("create");
        manager.start_vm("serial-test").expect("start");

        // Give the VM a moment to boot and print.
        tokio::time::sleep(std::time::Duration::from_secs(2)).await;

        let mut reader = manager
            .serial_reader("serial-test")
            .expect("serial reader");

        // Check if any serial output arrived.
        let chunk = tokio::time::timeout(
            std::time::Duration::from_secs(5),
            reader.read_chunk(),
        )
        .await;

        // Stop the VM.
        let _ = manager.stop_vm("serial-test");

        if let Ok(Some(data)) = chunk {
            let text = String::from_utf8_lossy(&data);
            assert!(
                text.contains("Hello") || !text.is_empty(),
                "expected serial output, got: {text}"
            );
        }
    });
}
