  173|       |/// A virtual processor — the vCPU run loop lives here.
  174|       |///
  175|       |/// # Abstract
  176|       |///
  177|       |/// Each vCPU usually runs on its own dedicated OS thread. The `run` method
  178|       |/// enters the guest execution context and blocks until the guest exits
  179|       |/// due to an event like I/O, MMIO, or HLT.
  180|       |///
  181|       |/// # The Hero's Journey
  182|       |///
  183|       |/// ```rust
  184|       |/// use hitz_hal::{Vcpu, HalError, VcpuExit};
  185|       |/// # struct DummyVcpu;
  186|       |/// # impl Vcpu for DummyVcpu {
  187|       |/// #     type CancelHandle = ();
  188|       |/// #     fn run(&mut self) -> Result<hitz_hal::VcpuExit, HalError> { Ok(VcpuExit::Halt) }
  189|       |/// #     fn cancel_handle(&self) -> () { () }
  190|       |/// #     fn cancel_via(_: &()) -> Result<(), HalError> { Ok(()) }
  191|       |/// #     fn get_regs(&self) -> Result<hitz_hal::StandardRegs, HalError> { unreachable!() }
  192|       |/// #     fn set_regs(&mut self, _: &hitz_hal::StandardRegs) -> Result<(), HalError> { Ok(()) }
  193|       |/// #     fn get_sregs(&self) -> Result<hitz_hal::SpecialRegs, HalError> { unreachable!() }
  194|       |/// #     fn set_sregs(&mut self, _: &hitz_hal::SpecialRegs) -> Result<(), HalError> { Ok(()) }
  195|       |/// #     fn inject_interrupt(&mut self, _: u8) -> Result<(), HalError> { Ok(()) }
  196|       |/// #     fn request_interrupt_window(&mut self) -> Result<(), HalError> { Ok(()) }
  197|       |/// # }
  198|       |/// # let mut vcpu = DummyVcpu;
  199|       |/// loop {
  200|       |///     match vcpu.run().expect("vcpu run failed") {
  201|       |///         VcpuExit::Halt => break, // Guest executed HLT
  202|       |///         _ => {} // Handle other exits
  203|       |///     }
  204|       |/// }
  205|       |/// ```
  206|       |///
  207|       |/// # Details
  208|       |///
  209|       |/// Call [`cancel_handle`][Vcpu::cancel_handle] *before* moving the vCPU into
  210|       |/// a thread. Share the handle via `Arc<Self::CancelHandle>` or by cloning,
  211|       |/// then call [`cancel_via`][Vcpu::cancel_via] from any thread to interrupt a
  212|       |/// blocked `run` call.
  213|       |pub trait Vcpu: Send {
  214|       |    /// Opaque handle that can cancel a running vCPU from any thread.
  215|       |    ///
  216|       |    /// Must be extracted before the vCPU is moved into a thread.
  217|       |    /// Safe to clone and share across threads via `Arc` or direct clone.
  218|       |    type CancelHandle: Send + Sync + Clone;
  219|       |
  220|       |    /// Enter the guest and run until an exit occurs.
  221|       |    fn run(&mut self) -> Result<VcpuExit, HalError>;
  222|       |
  223|       |    /// Extract a cancel handle for this vCPU.
  224|       |    ///
  225|       |    /// Call this *before* moving the vCPU into a thread. The returned
  226|       |    /// handle can be cloned and shared freely across threads.
  227|       |    fn cancel_handle(&self) -> Self::CancelHandle;
  228|       |
  229|       |    /// Cancel a running vCPU using a previously extracted handle.
  230|       |    ///
  231|       |    /// Forces `run` to return `VcpuExit::Canceled`. Safe to call from
  232|       |    /// any thread. Returns `Ok(())` if the cancel was delivered (the
  233|       |    /// vCPU may not have exited yet).
  234|       |    ///
  235|       |    /// Returns `Err` if the underlying hypervisor call fails (e.g. the
  236|       |    /// partition has already been destroyed).
  237|       |    fn cancel_via(handle: &Self::CancelHandle) -> Result<(), HalError>;
  238|       |
  239|       |    /// Cancel a running vCPU from another thread.
  240|       |    ///
  241|       |    /// This causes `run` to return `VcpuExit::Canceled`.
  242|      0|    fn cancel(&self) -> Result<(), HalError> {
  243|      0|        Self::cancel_via(&self.cancel_handle())
  244|      0|    }

  246|       |    /// Read the general-purpose registers.
  247|       |    fn get_regs(&self) -> Result<StandardRegs, HalError>;
  248|       |
  249|       |    /// Write the general-purpose registers.
  250|       |    fn set_regs(&mut self, regs: &StandardRegs) -> Result<(), HalError>;
  251|       |
  252|       |    /// Read the special (system) registers.
  253|       |    fn get_sregs(&self) -> Result<SpecialRegs, HalError>;
  254|       |
  255|       |    /// Write the special (system) registers.
  256|       |    fn set_sregs(&mut self, sregs: &SpecialRegs) -> Result<(), HalError>;
  257|       |
  258|       |    /// Inject an external interrupt into the vCPU.
  259|       |    ///
  260|       |    /// On WHP this sets `WHvRegisterPendingInterruption` before re-entering
  261|       |    /// the guest. The guest must have IF=1 (interrupts enabled) for this
  262|       |    /// to take effect immediately.
  263|       |    fn inject_interrupt(&mut self, vector: u8) -> Result<(), HalError>;
